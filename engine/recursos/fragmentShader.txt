#version 300 es
precision mediump float;

// Variables de entrada desde el vertex shader
in vec3 v_normal;
in vec3 v_position;

// Uniforms para iluminación
uniform vec3 u_lightPosition;
uniform vec3 u_lightColor;
uniform vec3 u_ambientColor;
uniform vec3 u_diffuseColor;

// Salida del fragment shader
out vec4 fragColor;

void main() {
    // Normalizar la normal
    vec3 normal = normalize(v_normal);
    
    // Calcular dirección de la luz
    vec3 lightDir = normalize(u_lightPosition - v_position);
    
    // Calcular intensidad difusa
    float diff = max(dot(normal, lightDir), 0.0);
    
    // Calcular color final
    vec3 ambient = u_ambientColor * u_diffuseColor;
    vec3 diffuse = diff * u_lightColor * u_diffuseColor;
    vec3 result = ambient + diffuse;
    
    // Asignar color final
    fragColor = vec4(result, 1.0);
}
